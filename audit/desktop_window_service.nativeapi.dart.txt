import 'package:flutter/foundation.dart';
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:path_provider/path_provider.dart';
import 'package:tray_manager/tray_manager.dart' as tray_manager;
import 'package:universal_io/io.dart';
import 'package:window_manager/window_manager.dart' as window_manager;
import 'package:windows_taskbar/windows_taskbar.dart';

/// Manages desktop window lifecycle, system tray, and taskbar alerts.
///
/// Window and tray control comes from the nativeapi-based window_manager and
/// tray_manager API: every call below is a synchronous FFI call, and events
/// arrive through listeners registered on the window, the tray icon and the
/// menu items themselves.
class DesktopWindowService {
  DesktopWindowService._();

  static final DesktopWindowService instance = DesktopWindowService._();

  static bool get isDesktop =>
      !kIsWeb &&
      (defaultTargetPlatform == TargetPlatform.windows ||
          defaultTargetPlatform == TargetPlatform.macOS ||
          defaultTargetPlatform == TargetPlatform.linux);

  static bool get isWindows =>
      !kIsWeb && defaultTargetPlatform == TargetPlatform.windows;

  bool _initialized = false;
  bool _isFlashing = false;

  /// Set while [quitApp] is closing the window for real, so the close-request
  /// handler lets the close through instead of hiding to the tray.
  bool _isQuitting = false;

  window_manager.Window? _window;

  /// Unsubscribes from the window, tray icon and menu item events; filled as
  /// the listeners are registered, drained by [quitApp].
  final List<void Function()> _detachEventListeners = <void Function()>[];

  /// Name of the app in the window title.
  static const String appTitle = 'NiosMess';

  /// Smallest window, frame included. Wide enough for settings to keep the
  /// list of sections beside the open one: the app rail (97), the list (300)
  /// and the section (560), plus what the frame takes.
  static const Size minimumWindowSize = Size(980, 560);

  String _title = appTitle;

  /// Initializes window bounds, constraints, system tray, and taskbar integration.
  Future<void> initialize() async {
    if (!isDesktop || _initialized) return;
    _initialized = true;

    try {
      final window = window_manager.WindowManager.instance.getCurrent();
      if (window == null) {
        throw StateError('no desktop window is available');
      }
      _window = window;

      window.title = appTitle;
      window.minimumSize = minimumWindowSize.toNative();
      window.show();
      window.focus();

      // Prevent closing directly so we can minimize to tray: the listener
      // below cancels the close request and hides the window instead.
      final windowListenerId = window.addListener(_onWindowEvent);
      _detachEventListeners.add(() => window.removeListener(windowListenerId));

      await _setupSystemTray();
    } catch (e) {
      debugPrint('[DesktopWindowService] Window initialization failed: $e');
    }
  }

  Future<void> _setupSystemTray() async {
    try {
      final trayIcon = tray_manager.TrayIcon.create();
      if (trayIcon == null) {
        throw StateError('no native tray icon is available');
      }

      // Extract ico to temp directory for reliable native Windows loading
      String iconPath;
      if (isWindows) {
        try {
          final ByteData data = await rootBundle.load('assets/app_icon.ico');
          final Directory tempDir = await getTemporaryDirectory();
          final File iconFile = File('${tempDir.path}/niosmess_tray.ico');
          await iconFile.writeAsBytes(data.buffer.asUint8List(), flush: true);
          iconPath = iconFile.path;
        } catch (_) {
          // Fallback to relative path if bundle loading failed
          iconPath = 'windows/runner/resources/app_icon.ico';
        }
      } else {
        iconPath = 'assets/NiosMess_icon.png';
      }

      final tray_manager.Image? icon = isWindows
          ? tray_manager.Image.fromFile(iconPath)
          : tray_manager.ImageAsset.fromAsset(iconPath);
      if (icon == null) {
        throw ArgumentError.value(
          iconPath,
          'iconPath',
          'Unable to load tray icon',
        );
      }

      trayIcon
        ..icon = icon
        ..setVisible(true)
        ..setTooltip('NiosMess');

      final menu = tray_manager.Menu.create();
      final showWindowItem = tray_manager.MenuItem.createWithLabelAndType(
        'Открыть NiosMess',
        tray_manager.MenuItemType.normal,
      );
      final exitItem = tray_manager.MenuItem.createWithLabelAndType(
        'Выйти',
        tray_manager.MenuItemType.normal,
      );
      if (menu == null || showWindowItem == null || exitItem == null) {
        throw StateError('no native tray menu is available');
      }

      // Menu item clicks arrive as events on the items themselves.
      final showWindowListenerId = showWindowItem.addListener(
        (tray_manager.MenuEvent event) {
          if (event is tray_manager.MenuItemClickedEvent) {
            showWindow();
          }
        },
      );
      _detachEventListeners.add(
        () => showWindowItem.removeListener(showWindowListenerId),
      );

      final exitListenerId = exitItem.addListener(
        (tray_manager.MenuEvent event) {
          if (event is tray_manager.MenuItemClickedEvent) {
            quitApp();
          }
        },
      );
      _detachEventListeners.add(() => exitItem.removeListener(exitListenerId));

      menu
        ..addItem(showWindowItem)
        ..addSeparator()
        ..addItem(exitItem);

      // Left click on the tray icon restores the window; right click opens
      // the context menu natively.
      trayIcon.setContextMenu(menu);
      trayIcon.setContextMenuTrigger(
        tray_manager.ContextMenuTrigger.rightClicked,
      );

      final trayListenerId = trayIcon.addListener(_onTrayIconEvent);
      _detachEventListeners.add(() => trayIcon.removeListener(trayListenerId));
    } catch (e) {
      debugPrint('[DesktopWindowService] Tray setup failed: $e');
    }
  }

  /// Shows [title] in the title bar. Does nothing before the window is set
  /// up, off the desktop, or when the title is already there.
  Future<void> setTitle(String title) async {
    if (!isDesktop || !_initialized || title == _title) return;
    _title = title;
    try {
      _window?.title = title;
    } catch (_) {}
  }

  /// Brings window to front and focuses it.
  Future<void> showWindow() async {
    if (!isDesktop) return;
    try {
      final window = _window;
      if (window == null) return;
      if (window.isMinimized) {
        window.restore();
      }
      window.show();
      window.focus();
      stopTaskbarFlashing();
    } catch (_) {}
  }

  /// Hides window to system tray.
  Future<void> hideWindow() async {
    if (!isDesktop) return;
    try {
      _window?.hide();
    } catch (_) {}
  }

  /// Quits application completely.
  Future<void> quitApp() async {
    if (!isDesktop) return;
    try {
      _isQuitting = true;

      for (final detach in _detachEventListeners.toList()) {
        try {
          detach();
        } catch (_) {}
      }
      _detachEventListeners.clear();

      // Closing the app's last window ends the app, as the platform's
      // runner decides; fall back to a hard exit when the window cannot.
      final window = _window;
      if (window == null || !window.close()) {
        exit(0);
      }
    } catch (_) {
      exit(0);
    }
  }

  /// Flashes the taskbar icon to alert user of an incoming message.
  Future<void> flashTaskbar() async {
    if (!isWindows || _isFlashing) return;
    try {
      final window = _window;
      if (window == null || window.isFocused) return;

      _isFlashing = true;
      await WindowsTaskbar.setFlashTaskbarAppIcon(
        mode: TaskbarFlashMode.all | TaskbarFlashMode.timernofg,
        flashCount: 5,
        timeout: const Duration(milliseconds: 500),
      );
    } catch (_) {}
  }

  /// Resets taskbar flashing once user returns or focuses the window.
  Future<void> stopTaskbarFlashing() async {
    if (!isWindows || !_isFlashing) return;
    _isFlashing = false;
    try {
      await WindowsTaskbar.resetFlashTaskbarAppIcon();
    } catch (_) {}
  }

  // ── Event handlers ─────────────────────────────────────────────────────────

  /// Routes window events: a close request is cancelled and the window is
  /// hidden to the tray (unless the app is quitting), focus stops taskbar
  /// flashing.
  Future<void> _onWindowEvent(window_manager.WindowEvent event) async {
    if (event is window_manager.WindowCloseRequestedEvent) {
      if (_isQuitting) return;
      // Default behavior on desktop: minimize to tray on X.
      event.request.cancel();
      await hideWindow();
    } else if (event is window_manager.WindowFocusedEvent) {
      stopTaskbarFlashing();
    }
  }

  /// Routes tray icon events: a click on the icon brings the window back.
  Future<void> _onTrayIconEvent(tray_manager.TrayIconEvent event) async {
    if (event is tray_manager.TrayIconClickedEvent) {
      await showWindow();
    }
  }
}
